// Normalize whatever the API returns into a LeaderboardUser[].
function normalizeUsers(data: unknown): LeaderboardUser[] {
  const raw = Array.isArray(data)
    ? data
    : Array.isArray((data as { users?: unknown[] })?.users)
      ? (data as { users: unknown[] }).users
      : [];

  return raw
    .map((item, index) => {
      const u = (item ?? {}) as Record<string, unknown>;
      return {
        id: String(u.id ?? u.username ?? index),
        name: String(u.name ?? u.username ?? "User"),
        avatarUrl:
          typeof u.avatarUrl === "string"
            ? u.avatarUrl
            : typeof u.avatar === "string"
              ? u.avatar
              : undefined,
        entries: Number(u.entries ?? u.totalEntries ?? 0),
        wins: Number(u.wins ?? u.totalWins ?? 0),
        points: Number(u.points ?? u.score ?? 0),
      } satisfies LeaderboardUser;
    })
    .filter((u) => u.name !== "");
}

function rankBadge(rank: number): string {
  if (rank === 1) return "bg-yellow-400 text-yellow-950";
  if (rank === 2) return "bg-gray-300 text-gray-800";
