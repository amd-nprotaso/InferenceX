```python
#!/usr/bin/env python3
"""
Validate a JSON file against a JSON Schema and print the validation errors.
"""

import json
import sys

from jsonschema import Draft202012Validator, ValidationError


def load_json(filepath: str) -> dict:
    """Load and parse a JSON file."""
    try:
        with open(filepath, "r", encoding="utf-8") as f:
            return json.load(f)
    except FileNotFoundError:
        print(f"Error: File not found: {filepath}")
        sys.exit(1)
    except json.JSONDecodeError as e:
        print(f"Error: Invalid JSON in {filepath}: {e}")
        sys.exit(1)


def validate(data: dict, schema: dict) -> list[ValidationError]:
    """
    Validate a JSON document against a JSON Schema.
    Returns a list of all validation errors.
    """
    validator = Draft202012Validator(schema)
    # iter_errors returns ALL errors, not just the first one
    return list(validator.iter_errors(data))


def format_error(error: ValidationError) -> str:
    """Format a single validation error into a readable string."""
    path = " -> ".join(str(p) for p in error.absolute_path) or "(root)"
    return f"  Path: {path}\n  Error: {error.message}\n"


def main():
    if len(sys.argv) != 3:
        print("Usage: python validate.py <data.json> <schema.json>")
        sys.exit(1)

    data_path = sys.argv[1]
    schema_path = sys.argv[2]

    # Load the data and the schema
    data = load_json(data_path)
    schema = load_json(schema_path)

    # Run the schema validation
    errors = validate(data, schema)

    if not errors:
        print("✅ Validation passed: the document is valid.")
    else:
        print(f"❌ Validation failed with {len(errors)} error(s):\n")
        for i, error in enumerate(errors, 1):
            print(f"  [{i}] {format_error(error)}")


if __name__ == "__main__":
    main()
```

import csv
import json
import sys
from pathlib import Path


# ──────────────────────────────────────────────
#  Helpers
# ──────────────────────────────────────────────

def coerce_value(value: str | None):
    """
    Try to convert a string to a more specific JSON type.
    - "" / None  →  None
    - "true" / "false" (case-insensitive)  →  bool
    - "123" / "123.0" / "1e5"  →  int / float
    - otherwise  →  str (unchanged)
    """
    if value is None or value == "":
        return None

    # Boolean
    if value.lower() in ("true", "true"):
        return True
    if value.lower() in ("false", "false"):
        return False

    # Try int first, then float
    try:
        return int(value)
    except ValueError:
        pass
    try:
        return float(value)
    except ValueError:
        pass

    return value


def build_headers(headers: list[str]) -> list[str]:
    """
    Deduplicate column names.
    "name", "name", "name" → ["name", "name_2", "name_3"]
    """
    seen: dict[str, int] = {}
    result: list[str] = []
    for h in headers:
        h = h.strip() or f"column_{len(result) + 1}"
        if h in seen:
            seen[h] += 1
            result.append(f"{h}_{seen[h]}")
        else:
            seen[h] = 1
            result.append(h)
    return result

#  Core logic
# ──────────────────────────────────────────────

    csv_path: str | Path,
    *,

    coerce: bool = True,

        # ── Header ────────────────────────────────────────────────

                raw_headers = next(reader)
            except StopIteration:
                raise ValueError("CSV file has a header but no data rows.")

#  CLI
# ──────────────────────────────────────────────

def find_max(numbers):
    """
    Return the maximum value from a list of integers.
    Uses a simple loop instead of the built-in max() function.
    """
    if not numbers:

    current_max = numbers[0]

    for num in numbers[1:]:
        if num > current_max:

    return current_max

|---|---|

| **3. Loop** | Walk through the remaining numbers. If a number is **greater than** `current_max`, update `current_max`. |
| **4. Return** | After the loop, `current_max` holds the largest value. |

## Examples

    validator = Draft202012Validator(schema)
    errors = sorted(

    else:

  • Missing / empty values  →  null

  • Special characters in headers (whitespace, duplicates)
  • UTF-8 BOM and other encoding issues
  • Empty rows
  • Non-existent files / empty files

Usage:
    python csv_to_json.py input.csv output.json [--no-header] [--no-convert]

import argparse
import csv
import json
import sys
from pathlib import Path


# ──────────────────────────────────────────────
#  Value conversion
# ──────────────────────────────────────────────

def convert_value(value: str) -> object:
    """
    Convert a raw CSV string to a more specific JSON type.

    "true" / "false"   →  bool
    "123"              →  int
    "3.14"             →  float
    ""                 →  None  (null)
    "hello"          →  "hello" (unchanged)
    """
    if value == "":
        return None

    # Boolean
    if value.lower() == "true":
        return True
    if value.lower() == "false":
        return False

    # Integer
    try:
        return int(value)
    except ValueError:
        pass

    # Float
    try:
        return float(value)
    except ValueError:
        pass

    return value


# ──────────────────────────────────────────────
#  Header handling
# ──────────────────────────────────────────────

def build_headers(raw_headers: list[str]) -> list[str]:
    """
    Clean up headers:
      • Strip surrounding whitespace
      • Replace empty names with "col_N"
      • Deduplicate (e.g. "name", "name", "name" → "name", "name_2", "name_3")
    """
    seen: dict[str, int] = {}
    result: list[str] = []

    for i, name in enumerate(raw_headers):
        name = name.strip() or f"col_{i + 1}"

        if name in seen:
            seen[name] += 1
            result.append(f"{name}_{seen[name]}")
        else:
            seen[name] = 1
            result.append(name)

    return result


# ──────────────────────────────────────────────
#  Core conversion
# ──────────────────────────────────────────────

def csv_to_json(
    input_path: str | Path,
    output_path: str | Path,
    *,
    has_header: bool = True,
    convert_types: bool = True,
    encoding: str = "utf-8-sig",   # handles BOM automatically
) -> int:
    """
    Read a CSV file and write a JSON array of objects to *output_path*.

    Returns the number of records written.
    """
    input_path = Path(input_path)
    output_path = Path(output_path)

    # ── Validate input ──────────────────────────────────────────
    if not input_path.exists():
        raise FileNotFoundError(f"Input file not found: {input_path}")

    if input_path.stat().st_size == 0:
        raise ValueError(f"Input file is empty: {input_path}")

    # ── Read CSV ────────────────────────────────────────────────
    with open(input_path, newline="", encoding=encoding) as f:
        # Try to detect dialect; fall back to default
        try:
            sample = f.read(4096)
            f.seek(0)
            dialect = csv.Sniffer().sniff(sample)
        except csv.Error:
            dialect = csv.excel

        reader = csv.reader(f, dialect)

        # ── Header ──────────────────────────────────────────────
        if has_header:
            try:
                raw_headers = next(reader)
            except StopIteration:
                raise ValueError("CSV file has no data rows.")
            headers = build_headers(raw_headers)
        else:
            # Peek at first row to determine column count
            try:
                first_row = next(reader)
            except StopIteration:
                raise ValueError("CSV file is empty.")
            headers = [f"col_{i + 1}" for i in range(len(first_row))]
            # Prepend the peeked row back
            reader = [first_row] + reader

        # ── Build records ───────────────────────────────────────
        records: list[dict] = []
        for row in reader:
            # Skip completely empty rows
            if not any(row):
                continue

            # Pad / truncate to match header length
            if len(row) < len(headers):
                row = row + [""] * (len(headers) - len(row))
            elif len(row) > len(headers):
                print(
                    f"  ⚠  Row has {len(row)} cols "
                    f"(expected {len(headers)}); extra cols ignored.",
                    file=sys.stderr,
                )

            record = {}
            for header, value in zip(headers, row):
                record[header] = convert_value(value) if convert_types else value

            records.append(record)

    # ── Write JSON ──────────────────────────────────────────────
    with open(output_path, "w", encoding="utf-8") as f:
        json.dump(records, f, ensure_ascii=False, indent=2)

    return len(records)


# ──────────────────────────────────────────────
#  CLI
# ──────────────────────────────────────────────

def main():
    parser = argparse.ArgumentParser(
        description="Convert a CSV file to a JSON array file."
    )
    parser.add_argument("input_csv",   help="Path to the input CSV file")
    parser.add_argument("output_json", help="Path to the output JSON file")
    parser.add_argument(
        "--no-header", action="store_true",
        help="CSV has no header row (columns auto-named col_1, col_2, …)",
    )
    parser.add_argument(
        "--no-convert", action="store_true",

        Error:  'abc' is not of type 'integer'

"""
json_validate.py — Validate a JSON file against a JSON Schema.

Usage:
    python json_validate.py <data.json> <schema.json>

Dependencies:
    pip install jsonschema
"""

import json
import sys

from jsonschema import Draft202012Validator, ValidationError
from jsonschema.exceptions import SchemaError


# ──────────────────────────────────────────────
#  Helpers
# ──────────────────────────────────────────────

def load_json(filepath: str) -> dict | list:
    """Load and parse a JSON file with clear error messages."""
    try:
        with open(filepath, "r", encoding="utf-8") as f:
            return json.load(f)
    except FileNotFoundError:
        print(f"❌  File not found: {filepath}")
        sys.exit(1)
    except json.JSONDecodeError as e:
        print(f"❌  Invalid JSON in '{filepath}': {e}")
        sys.exit(1)


def format_error(error: ValidationError) -> str:
    """
    Format a single validation error into a readable string.

    Example output:
        Path:   user.name -> first
        Error:  'abc' is not of type 'integer'
    """
    path = " -> ".join(str(p) for p in error.absolute_path) or "(root)"
    return f"  Path:   {path}\n  Error:  {error.message}\n"


# ──────────────────────────────────────────────
#  Core validation
# ──────────────────────────────────────────────

def validate(data, schema) -> list[ValidationError]:
    """
    Validate a JSON document against a JSON Schema.

    Returns a list of ALL validation errors (not just the first one).
    """
    # 1. Ensure the schema itself is valid
    try:
        Draft202012Validator.check_schema(schema)
    except SchemaError as e:
        print(f"❌  The JSON Schema is invalid: {e.message}")
        sys.exit(1)

    # 2. Collect ALL errors (iter_errors ≠ stop at first)
    validator = Draft202012Validator(schema)
    errors = sorted(
        validator.iter_errors(data),
        key=lambda e: list(e.absolute_path),
    )
    return errors


# ──────────────────────────────────────────────
#  CLI
# ──────────────────────────────────────────────

def main():
    if len(sys.argv) != 3:
        print("Usage: python json_validate.py <data.json> <schema.json>")
        sys.exit(1)

    data_path   = sys.argv[1]
    schema_path = sys.argv[2]

    # Load both files
    data   = load_json(data_path)
    schema = load_json(schema_path)

    # ── Validate ──────────────────────────────────────────────
    errors = validate(data, schema)

    if not errors:
        print("✅  Validation passed: the document is valid.")
    else:
        print(f"❌  Validation failed with {len(errors)} error(s):\n")
        for i, error in enumerate(errors, 1):
            print(f"  [{i}] {format_error(error)}")
        sys.exit(1)


if __name__ == "__main__":
    main()

#!/usr/bin/env python3
"""
csv_to_json.py — Convert a CSV file to a JSON file.

Handles edge cases:
  • Missing / empty values  →  null
  • Numeric / boolean values →  auto-converted to proper types
  • Special characters in headers (whitespace, duplicates)
  • UTF-8 BOM and other encoding issues
  • Empty rows
  • Non-existent / empty files

Usage:
    python csv_to_json.py input.csv output.json [--no-header] [--no-convert]
"""

import argparse
import csv
import json
import sys
from pathlib import Path


# ──────────────────────────────────────────────
#  Value conversion
# ──────────────────────────────────────────────

def convert_value(value: str) -> object:
    """
    Convert a raw CSV string to a more specific JSON type.

    "true" / "false"   →  bool
    "123"              →  int
    "3.14"             →  float
    ""                 →  None  (null)
    "1e5"              →  100000.0
    "hello"            →  "hello" (unchanged)
    """
    if value == "":
        return None

    # Boolean
    if value.lower() == "true":
        return True
    if value.lower() == "false":
        return False

    # Integer
    try:
        return int(value)
    except ValueError:
        pass

    # Float (including scientific notation like "1e5")
    try:
        return float(value)
    except ValueError:
        pass

    return value


# ──────────────────────────────────────────────
#  Header handling
# ──────────────────────────────────────────────

def build_headers(raw_headers: list[str]) -> list[str]:
    """
    Clean up headers:
      • Strip surrounding whitespace
      • Replace empty names with "col_N"
      • Deduplicate (e.g. "name", "name", "name" → "name", "name_2", "name_3")
    """
    seen: dict[str, int] = {}
    result: list[str] = []

    for i, name in enumerate(raw_headers):
        name = name.strip() or f"col_{i + 1}"

        if name in seen:
            seen[name] += 1
            result.append(f"{name}_{seen[name]}")
        else:
            seen[name] = 1
            result.append(name)

    return result


# ──────────────────────────────────────────────
#  Core conversion
# ──────────────────────────────────────────────

def csv_to_json(
    input_path: str | Path,
    output_path: str | Path,
    *,
    has_header: bool = True,
    convert_types: bool = True,
    encoding: str = "utf-8-sig",   # handles BOM automatically
) -> int:
    """
    Read a CSV file and write a JSON array of objects to *output_path*.

    Returns the number of records written.
    """
    input_path = Path(input_path)
    output_path = Path(output_path)

    # ── Validate input ──────────────────────────────────────────
    if not input_path.exists():
        raise FileNotFoundError(f"Input file not found: {input_path}")

    if input_path.stat().st_size == 0:
        raise ValueError(f"Input file is empty: {input_path}")

    # ── Read CSV ────────────────────────────────────────────────
    with open(input_path, newline="", encoding=encoding) as f:
        # Try to detect dialect; fall back to default
        try:
            sample = f.read(4096)
            f.seek(0)
            dialect = csv.Sniffer().sniff(sample)
        except csv.Error:
            dialect = csv.excel

        reader = csv.reader(f, dialect)

        # ── Header ──────────────────────────────────────────────
        if has_header:
            try:
                raw_headers = next(reader)
            except StopIteration:
                raise ValueError("CSV file has no data rows.")
            headers = build_headers(raw_headers)
        else:
            # Peek at first row to determine column count
            try:
                first_row = next(reader)
            except StopIteration:
                raise ValueError("CSV file is empty.")
            headers = [f"col_{i + 1}" for i in range(len(first_row))]
            # Prepend the peeked row back
            reader = [first_row] + reader

        # ── Build records ───────────────────────────────────────
        records: list[dict] = []
        for row in reader:
            # Skip completely empty rows
            if not any(row):
                continue

            # Pad / truncate to match header length
            if len(row) < len(headers):
                row = row + [""] * (len(headers) - len(row))
            elif len(row) > len(headers):
                print(
                    f"  ⚠  Row has {len(row)} cols "
                    f"(expected {len(headers)}); extra cols ignored.",
                    file=sys.stderr,
                )

            record = {}
            for header, value in zip(headers, row):
                record[header] = convert_value(value) if convert_types else value

            records.append(record)

    # ── Write JSON ──────────────────────────────────────────────
    with open(output_path, "w", encoding="utf-8") as f:
        json.dump(records, f, ensure_ascii=False, indent=2)

    return len(records)


# ──────────────────────────────────────────────
#  CLI
# ──────────────────────────────────────────────

def main():
    parser = argparse.ArgumentParser(
        description="Convert a CSV file to a JSON array file."
    )
    parser.add_argument("input_csv",   help="Path to the input CSV file")
    parser.add_argument("output_json", help="Path to the output JSON file")
    parser.add_argument(
        "--no-header", action="store_true",
        help="CSV has no header row (columns auto-named col_1, col_2, …)",
    )
    parser.add_argument(
        "--no-convert", action="store_true",
        help="Keep all values as strings (skip numeric/boolean conversion)",
    )
    args = parser.parse_args()

    try:
        count = csv_to_json(
            args.input_csv,
            args.output_json,
            has_header=not args.no_header,
            convert_types=not args.no_convert,
        )
    except (FileNotFoundError, ValueError) as exc:
        print(f"Error: {exc}", file=sys.stderr)
        sys.exit(1)

    print(f"✅  {count} record(s) → {args.output_json}")


if __name__ == "__main__":
    main()
